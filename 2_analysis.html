build_pid_lodes <- function(df_input) {
  labs <- party_names

  from <- df_input |>
    select(id_w9, starts_with("party_id_w9")) |>
    pivot_longer(-id_w9, names_to = "var", values_to = "val") |>
    filter(val == 1) |>
    transmute(
      id_w9,
      from_party = recode(
        sub("^party_id_w9_(.*)$", "\\1", var),
        !!!labs
      )
    )

  to <- df_input |>
    select(id_w9, starts_with("party_id_w10")) |>
    pivot_longer(-id_w9, names_to = "var", values_to = "val") |>
    filter(val == 1) |>
    transmute(
      id_w9,
      to_party = recode(
        sub("^party_id_w10_(.*)$", "\\1", var),
        !!!labs
      )
    )

  transitions_indiv <- inner_join(from, to, by = "id_w9")

  transitions <- transitions_indiv |>
    count(`Wave 9` = from_party, `Wave 10` = to_party, name = "Freq")

  totN <- sum(transitions$Freq)

  share_w9 <- transitions |>
    group_by(`Wave 9`) |>
    summarise(s9 = sum(Freq) / totN, .groups = "drop")

  share_w10 <- transitions |>
    group_by(`Wave 10`) |>
    summarise(s10 = sum(Freq) / totN, .groups = "drop")

  net_pp <- share_w10 |>
    left_join(share_w9, by = c("Wave 10" = "Wave 9")) |>
    mutate(
      delta_pp = (s10 - s9) * 100,
      label_right = sprintf("%s\n(%+0.2f pp)", `Wave 10`, delta_pp)
    ) |>
    select(`Wave 10`, label_right)

  lodes <- transitions |>
    mutate(Share = Freq / totN) |>
    to_lodes_form(axes = 1:2, id = "alluvium") |>
    mutate(
      x = factor(x, levels = c("Wave 9", "Wave 10")),
      stratum_lab = ifelse(
        x == "Wave 10",
        net_pp$label_right[match(stratum, net_pp$`Wave 10`)],
        as.character(stratum)
      )
    )

  party_cols <- c(
    "CDU/CSU" = "#000000",
    "SPD" = "#E3000F",
    "Grüne" = "#0DB02B",
    "FDP" = "#FFED00",
    "Linke" = "#E2382A",
    "AfD" = "#009EE0",
    "BSW" = "#18A19A"
  )

  list(lodes = lodes, party_cols = party_cols)
}

# Build correlation data for a single wave and treatment status
build_cor_df <- function(df, wave_label) {
  groups <- list(
    list(treat = "Treated", data = df |> filter(workshop_Treatment == 1)),
    list(treat = "Control", data = df |> filter(workshop_Treatment == 0))
  )
  purrr::map_df(groups, function(x) {
    df_sel <- x$data |> select(all_of(v_outcomes))
    colnames(df_sel) <- names(v_outcomes)
    cmat <- cor(df_sel, use = "pairwise.complete.obs")
    as.data.frame(as.table(cmat)) |>
      rename(var1 = Var1, var2 = Var2, r = Freq) |>
      mutate(treat = x$treat, wave = wave_label)
  })
}

# Lin (2013) party interactions with demeaned dummies, video FE, cluster ID.
# Returns tidied bare ATE (party = "All parties") plus party-specific TEs.
# party_levels = display names; party_code_map = named codes (names = display).
# ref_party is omitted from the dummy set (collinearity); TE_ref recovered via contrasts.
fit_lin_party_feols <- function(df, outcome_var, party_levels, party_code_map = NULL,
                                ref_party = party_levels[length(party_levels)]) {
  if (is.null(party_code_map)) {
    party_code_map <- setNames(make.names(party_levels), party_levels)
  }
  interact_parties <- setdiff(party_levels, ref_party)
  stopifnot(length(interact_parties) >= 1)

  df_m <- df |>
    filter(!is.na(.data[[outcome_var]]), !is.na(party), party %in% party_levels) |>
    mutate(party = factor(party, levels = party_levels))

  # Raw exclusive dummies and sample shares (pre-demean); columns use stable codes
  pi <- setNames(numeric(length(interact_parties)), interact_parties)
  for (p in interact_parties) {
    raw <- as.integer(df_m$party == p)
    pi[p] <- mean(raw, na.rm = TRUE)
    df_m[[paste0("pdm_", party_code_map[[p]])]] <- raw - pi[p]
  }

  pdm_terms <- paste0("pdm_", party_code_map[interact_parties])
  rhs <- paste0(
    "workshop_Treatment * (",
    paste(pdm_terms, collapse = " + "),
    ")"
  )
  fit <- feols(
    as.formula(paste(outcome_var, "~", rhs, "| video")),
    data = df_m,
    cluster = ~ID
  )

  b <- coef(fit)
  V <- vcov(fit)
  treat <- "workshop_Treatment"
  if (!treat %in% names(b)) {
    stop("Treatment coefficient missing from Lin fit for ", outcome_var)
  }

  # Linear combination c'b with SE sqrt(c'Vc)
  lincom <- function(cvec) {
    cvec <- cvec[names(b)]
    cvec[is.na(cvec)] <- 0
    est <- sum(cvec * b)
    se <- sqrt(as.numeric(t(cvec) %*% V %*% cvec))
    tibble(
      estimate = est,
      std.error = se,
      conf.low = est - 1.96 * se,
      conf.high = est + 1.96 * se,
      p.value = 2 * pnorm(-abs(est / se))
    )
  }

  int_name <- function(p) paste0(treat, ":pdm_", party_code_map[[p]])

  # Bare treatment = Lin ATE under demeaning
  c_ate <- setNames(rep(0, length(b)), names(b))
  c_ate[treat] <- 1
  out_ate <- lincom(c_ate) |>
    mutate(term = treat, party = "All parties")

  # Party-specific TE = β + δ'(d_p - π)
  out_parties <- lapply(party_levels, function(p) {
    cvec <- setNames(rep(0, length(b)), names(b))
    cvec[treat] <- 1
    for (j in interact_parties) {
      nm <- int_name(j)
      if (!nm %in% names(b)) next
      if (identical(p, j)) {
        cvec[nm] <- 1 - pi[j]
      } else {
        cvec[nm] <- -pi[j]
      }
    }
    lincom(cvec) |>
      mutate(term = paste0("TE_", party_code_map[[p]]), party = p)
  }) |>
    bind_rows()

  bind_rows(out_ate, out_parties) |>
    mutate(outcome_var = outcome_var)
}

# Dropbox / multi-session Windows often locks *.rds mid-write ("cannot open the connection").
# Write to a same-directory temp file, then replace (rename-aside when target exists);
# retry; name the path on failure.
safe_write_rds <- function(object, path, max_tries = 5, sleep_sec = 0.5) {
  dir.create(dirname(path), showWarnings = FALSE, recursive = TRUE)
  abs_path <- normalizePath(path, winslash = "/", mustWork = FALSE)
  tmp <- file.path(
    dirname(path),
    paste0(".", basename(path), ".", Sys.getpid(), ".tmp")
  )
  bak <- paste0(path, ".", Sys.getpid(), ".bak")
  on.exit({
    if (file.exists(tmp)) unlink(tmp, force = TRUE)
    if (file.exists(bak)) unlink(bak, force = TRUE)
  }, add = TRUE)

  replace_with_tmp <- function() {
    if (!file.exists(tmp)) {
      stop("temp RDS missing after write_rds", call. = FALSE)
    }
    if (file.exists(path)) {
      # Prefer rename-aside then move: Dropbox often blocks overwrite while
      # still allowing rename (same pattern as safe_ggsave_pdf).
      if (file.exists(bak)) unlink(bak, force = TRUE)
      if (!isTRUE(file.rename(path, bak))) {
        copied <- isTRUE(file.copy(tmp, path, overwrite = TRUE))
        if (!copied) {
          stop("could not rename RDS aside or overwrite it", call. = FALSE)
        }
        unlink(tmp, force = TRUE)
        return(invisible(TRUE))
      }
      if (!isTRUE(file.rename(tmp, path))) {
        file.rename(bak, path)
        stop("could not move temp RDS into place after rename-aside", call. = FALSE)
      }
      unlink(bak, force = TRUE)
    } else if (!isTRUE(file.rename(tmp, path))) {
      if (!isTRUE(file.copy(tmp, path, overwrite = TRUE))) {
        stop("file.rename and file.copy both failed for RDS", call. = FALSE)
      }
      unlink(tmp, force = TRUE)
    }
    invisible(TRUE)
  }

  last_msg <- NULL
  for (i in seq_len(max_tries)) {
    ok <- tryCatch({
      if (file.exists(tmp)) unlink(tmp, force = TRUE)
      write_rds(object, tmp)
      replace_with_tmp()
      TRUE
    }, error = function(e) {
      last_msg <<- conditionMessage(e)
      if (file.exists(tmp)) unlink(tmp, force = TRUE)
      Sys.sleep(sleep_sec * i)
      FALSE
    })
    if (isTRUE(ok)) return(invisible(path))
  }

  stop(
    sprintf(
      paste0(
        "Could not write RDS to '%s' after %d tries (%s). ",
        "Usually a Dropbox sync lock or another RStudio/rsession holding the file. ",
        "Pause Dropbox sync and close other R sessions, then re-render."
      ),
      abs_path, max_tries,
      if (is.null(last_msg)) "unknown error" else last_msg
    ),
    call. = FALSE
  )
}

# Table chunks previously saved full fixest lists (scores / residuals / FE vectors
# → tens–hundreds of MB). Nothing in this qmd re-reads those RDS files (texreg
# uses in-memory fits). Persist tidy/glance trees instead via safe_write_rds.
slim_fixest_for_rds <- function(x) {
  if (inherits(x, "fixest")) {
    return(list(
      tidy = broom::tidy(x, conf.int = TRUE),
      glance = broom::glance(x),
      coeftable = as.data.frame(coeftable(x)),
      nobs = nobs(x),
      formula = as.character(formula(x))
    ))
  }
  if (is.list(x)) {
    return(lapply(x, slim_fixest_for_rds))
  }
  x
}

write_tab_rds <- function(models, path) {
  safe_write_rds(slim_fixest_for_rds(models), path)
}

# Same pattern for PDFs: ggsave opens the path with device(), which fails when a
# viewer / Dropbox / another session holds the existing file. Write to a
# same-directory temp PDF, then replace; on lock, rename the target aside first.
safe_ggsave_pdf <- function(plot, path, device, width = NA, height = NA,
                            max_tries = 5, sleep_sec = 0.5) {
  dir.create(dirname(path), showWarnings = FALSE, recursive = TRUE)
  abs_path <- normalizePath(path, winslash = "/", mustWork = FALSE)
  tmp <- file.path(
    dirname(path),
    paste0(".", basename(path), ".", Sys.getpid(), ".tmp.pdf")
  )
  bak <- paste0(path, ".", Sys.getpid(), ".bak")
  on.exit({
    if (file.exists(tmp)) unlink(tmp, force = TRUE)
    if (file.exists(bak)) unlink(bak, force = TRUE)
  }, add = TRUE)

  ggsave_to <- function(out, dev) {
    save_args <- list(filename = out, plot = plot, device = dev)
    if (!is.na(width)) save_args$width <- width
    if (!is.na(height)) save_args$height <- height
    do.call(ggsave, save_args)
  }

  replace_with_tmp <- function() {
    if (!file.exists(tmp)) {
      stop("temp PDF missing after ggsave", call. = FALSE)
    }
    if (file.exists(path)) {
      # Prefer rename-aside then move: Dropbox / PDF viewers often block
      # overwrite (file.copy / CreateFile) while still allowing rename.
      if (file.exists(bak)) unlink(bak, force = TRUE)
      if (!isTRUE(file.rename(path, bak))) {
        # Fall back to overwrite copy if rename-aside fails
        copied <- isTRUE(file.copy(tmp, path, overwrite = TRUE))
        if (!copied) {
          stop("could not rename PDF aside or overwrite it", call. = FALSE)
        }
        unlink(tmp, force = TRUE)
        return(invisible(TRUE))
      }
      if (!isTRUE(file.rename(tmp, path))) {
        # Restore prior PDF if move failed
        file.rename(bak, path)
        stop("could not move temp PDF into place after rename-aside", call. = FALSE)
      }
      unlink(bak, force = TRUE)
    } else if (!isTRUE(file.rename(tmp, path))) {
      if (!isTRUE(file.copy(tmp, path, overwrite = TRUE))) {
        stop("file.rename and file.copy both failed for PDF", call. = FALSE)
      }
      unlink(tmp, force = TRUE)
    }
    invisible(TRUE)
  }

  last_msg <- NULL
  use_device <- device
  for (i in seq_len(max_tries)) {
    ok <- tryCatch({
      if (file.exists(tmp)) unlink(tmp, force = TRUE)
      tryCatch(
        ggsave_to(tmp, use_device),
        error = function(e) {
          # cairo_pdf can fail on some Windows setups; fall back once to pdf()
          if (!identical(use_device, grDevices::pdf)) {
            warning(
              sprintf(
                "ggsave failed for temp PDF near '%s' (%s); falling back to pdf().",
                abs_path, conditionMessage(e)
              ),
              call. = FALSE
            )
            use_device <<- grDevices::pdf
            ggsave_to(tmp, use_device)
          } else {
            stop(e)
          }
        }
      )
      # If destination still locked, clear it before replace
      if (file.exists(path)) {
        can_write <- tryCatch({
          con <- file(path, open = "r+b")
          close(con)
          TRUE
        }, error = function(e) FALSE)
        if (!isTRUE(can_write)) {
          if (file.exists(bak)) unlink(bak, force = TRUE)
          if (!isTRUE(file.rename(path, bak))) {
            unlink(path, force = TRUE)
          }
        }
      }
      replace_with_tmp()
      TRUE
    }, error = function(e) {
      last_msg <<- conditionMessage(e)
      if (file.exists(tmp)) unlink(tmp, force = TRUE)
      Sys.sleep(sleep_sec * i)
      FALSE
    })
    if (isTRUE(ok)) return(invisible(path))
  }

  stop(
    sprintf(
      paste0(
        "Could not write PDF to '%s' after %d tries (%s). ",
        "Usually a PDF viewer (PDF24/Acrobat/Sumatra/Edge), Dropbox sync, ",
        "Explorer preview, or another R session holding the file. ",
        "Close the PDF if open, pause Dropbox, then re-render."
      ),
      abs_path, max_tries,
      if (is.null(last_msg)) "unknown error" else last_msg
    ),
    call. = FALSE
  )
}

write_fig <- function(stem, estimates, plot, width = NA, height = NA,
                      device = NULL, publish = NULL) {
  ensure_fig_dirs()
  dir.create(tab_dir, showWarnings = FALSE)
  est_path <- file.path(est_dir, paste0(stem, ".rds"))
  gg_path  <- file.path(gg_dir, paste0(stem, ".rds"))
  pdf_path <- file.path(fig_dir, paste0(stem, ".pdf"))

  safe_write_rds(estimates, est_path)

  # ggplot RDS is for later reuse; do not abort the render if Dropbox locks it
  tryCatch(
    safe_write_rds(plot, gg_path),
    error = function(e) {
      warning(
        sprintf(
          "Skipping ggplot RDS for stem '%s' at '%s' (%s). Estimates + PDF still written.",
          stem,
          normalizePath(gg_path, winslash = "/", mustWork = FALSE),
          conditionMessage(e)
        ),
        call. = FALSE
      )
    }
  )

  # cairo_pdf preferred when available; fall back to pdf() on Windows/device failure
  if (is.null(device)) {
    device <- if (isTRUE(capabilities("cairo"))) grDevices::cairo_pdf else grDevices::pdf
  }
  safe_ggsave_pdf(
    plot = plot,
    path = pdf_path,
    device = device,
    width = width,
    height = height
  )
  print(plot)
  if (!is.null(publish)) {
    publish_file(pdf_path, publish)
  }
  invisible(plot)
}

publish_file <- function(src, dest) {
  dir.create(dirname(dest), showWarnings = FALSE, recursive = TRUE)
  ok <- isTRUE(file.copy(src, dest, overwrite = TRUE))
  if (!ok) warning("Could not copy ", src, " -> ", dest)
  invisible(ok)
}

pair_cor <- function(x, y) {
  ok <- is.finite(x) & is.finite(y)
  n <- sum(ok)
  if (n < 3) return(list(r = NA_real_, n = n))
  list(r = suppressWarnings(cor(x[ok], y[ok])), n = n)
}
cluster_cor_diff_p <- function(yi, yj, mod, id) {
  ok <- is.finite(yi) & is.finite(yj) & is.finite(mod) & !is.na(id)
  if (sum(ok) < 8L) return(NA_real_)
  mod_ok <- as.numeric(mod[ok])
  if (length(unique(mod_ok)) < 2L) return(NA_real_)
  yi_ok <- as.numeric(yi[ok])
  yj_ok <- as.numeric(yj[ok])
  s_i <- stats::sd(yi_ok)
  s_j <- stats::sd(yj_ok)
  if (!is.finite(s_i) || !is.finite(s_j) || s_i <= 0 || s_j <= 0) return(NA_real_)
  d <- data.frame(
    yi_z = as.numeric(scale(yi_ok)),
    yj_z = as.numeric(scale(yj_ok)),
    mod = mod_ok,
    ID = as.character(id[ok]),
    stringsAsFactors = FALSE
  )
  fit <- tryCatch(feols(yi_z ~ yj_z * mod, data = d, cluster = ~ID), error = function(e) NULL)
  if (is.null(fit)) return(NA_real_)
  td <- broom::tidy(fit)
  hit <- td$term[grepl(":", td$term, fixed = TRUE)]
  if (length(hit) != 1L) return(NA_real_)
  as.numeric(td$p.value[td$term == hit])
}

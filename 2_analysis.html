# Party identification transition helpers
build_pid_lodes <- function(df_input) {
  labs <- party_names

  from <- df_input |>
    select(id_w9, starts_with("party_id_w9")) |>
    pivot_longer(-id_w9, names_to = "var", values_to = "val") |>
    filter(val == 1) |>
    transmute(
      id_w9,
      from_party = recode(
        sub("^party_id_w9_(.*)$", "\\1", var),
        !!!labs
      )
    )

  to <- df_input |>
    select(id_w9, starts_with("party_id_w10")) |>
    pivot_longer(-id_w9, names_to = "var", values_to = "val") |>
    filter(val == 1) |>
    transmute(
      id_w9,
      to_party = recode(
        sub("^party_id_w10_(.*)$", "\\1", var),
        !!!labs
      )
    )

  transitions_indiv <- inner_join(from, to, by = "id_w9")

  transitions <- transitions_indiv |>
    count(`Wave 9` = from_party, `Wave 10` = to_party, name = "Freq")

  totN <- sum(transitions$Freq)

  share_w9 <- transitions |>
    group_by(`Wave 9`) |>
    summarise(s9 = sum(Freq) / totN, .groups = "drop")

  share_w10 <- transitions |>
    group_by(`Wave 10`) |>
    summarise(s10 = sum(Freq) / totN, .groups = "drop")

  net_pp <- share_w10 |>
    left_join(share_w9, by = c("Wave 10" = "Wave 9")) |>
    mutate(
      delta_pp = (s10 - s9) * 100,
      label_right = sprintf("%s\n(%+0.2f pp)", `Wave 10`, delta_pp)
    ) |>
    select(`Wave 10`, label_right)

  lodes <- transitions |>
    mutate(Share = Freq / totN) |>
    to_lodes_form(axes = 1:2, id = "alluvium") |>
    mutate(
      x = factor(x, levels = c("Wave 9", "Wave 10")),
      stratum_lab = ifelse(
        x == "Wave 10",
        net_pp$label_right[match(stratum, net_pp$`Wave 10`)],
        as.character(stratum)
      )
    )

  party_cols <- c(
    "CDU/CSU" = "#000000",
    "SPD" = "#E3000F",
    "Grüne" = "#0DB02B",
    "FDP" = "#FFED00",
    "Linke" = "#E2382A",
    "AfD" = "#009EE0",
    "BSW" = "#18A19A"
  )

  list(lodes = lodes, party_cols = party_cols)
}

# Build correlation data for a single wave and treatment status
build_cor_df <- function(df, wave_label) {
  groups <- list(
    list(treat = "Treated", data = df |> filter(workshop_Treatment == 1)),
    list(treat = "Control", data = df |> filter(workshop_Treatment == 0))
  )
  purrr::map_df(groups, function(x) {
    df_sel <- x$data |> select(all_of(v_outcomes))
    colnames(df_sel) <- names(v_outcomes)
    cmat <- cor(df_sel, use = "pairwise.complete.obs")
    as.data.frame(as.table(cmat)) |>
      rename(var1 = Var1, var2 = Var2, r = Freq) |>
      mutate(treat = x$treat, wave = wave_label)
  })
}


# Main anlaysis function clusters by ID and fixed effects by video
# Run feols by group and return tidied coefficients with metadata
run_feols_tidy <- function(df, outcome_var, extra_cols = list()) {
  feols(
    as.formula(paste(outcome_var, "~ workshop_Treatment | video")),
    data = df,
    cluster = ~ID
  ) |>
    tidy(conf.int = TRUE) |>
    mutate(!!!extra_cols)
}

# Persist estimates, ggplot object, and PDF under matching stems
ensure_fig_dirs <- function() {
  dir.create(est_dir, showWarnings = FALSE)
  dir.create(gg_dir, showWarnings = FALSE)
  dir.create(fig_dir, showWarnings = FALSE)
}

write_fig <- function(stem, estimates, plot, width = NA, height = NA,
                      device = NULL) {
  ensure_fig_dirs()
  write_rds(estimates, file.path(est_dir, paste0(stem, ".rds")))
  write_rds(plot, file.path(gg_dir, paste0(stem, ".rds")))
  # cairo_pdf is preferred when available; plain pdf works on Windows without Cairo
  if (is.null(device)) {
    device <- if (isTRUE(capabilities("cairo"))) grDevices::cairo_pdf else grDevices::pdf
  }
  save_args <- list(
    filename = file.path(fig_dir, paste0(stem, ".pdf")),
    plot = plot,
    device = device
  )
  if (!is.na(width)) save_args$width <- width
  if (!is.na(height)) save_args$height <- height
  do.call(ggsave, save_args)
  print(plot)
  invisible(plot)
}

Comparing video types

Formal tests of whether short-term treatment effects on liking and reporting differ across video types. For each outcome and group we estimate

\[ Y = \beta_0 + \beta_1 \text{Treatment} + \beta_2 \text{Type} + \beta_3 (\text{Treatment} \times \text{Type}) + \varepsilon \]

on the two types being compared only (no video fixed effects, so all three slopes are identified). Type = 1 for emotionally manipulative political content and 0 for the comparison type. Standard errors are clustered by respondent. \(\beta_1\) is the ATE for the comparison type; \(\beta_3\) is the difference in ATEs (emotional minus comparison).

Groups: ALL (pooled) and each sending party. Short-term sample (wave_9) only. Outcomes: Like and Report. One regression table per outcome; columns are groups.

  y <- v_outcomes[[outcome_name]]

Exploratory: Causal-forest CATE correlations across outcomes

Exploratory appendix analysis (not a primary confirmatory claim). We estimate conditional average treatment effects (CATEs) separately for Like, Share, Bookmark, and Report using grf::causal_forest, then inspect how those CATEs co-vary across outcomes.

Fitting four causal forests is slow. The switch RUN_FORESTS in the configuration chunk controls whether forests are refit (TRUE) or whether the document reuses saved CATEs from ests/fig_causal_forest_cate_corr.rds (FALSE, default) and only rebuilds the correlation grid / figure. Set RUN_FORESTS <- TRUE once after a code or data change that affects the forests, then return it to FALSE for ordinary compiles.

Unit of analysis. Each observation is a person × sending-party cell (ID × party): on emotionally manipulative political videos (strategy_2 == strategies[1] / "political\n(emotional)", wave_9), outcomes are the mean of the binary engagement indicators over videos from that party shown to that respondent. A unique unit id pp_id identifies each person–party cell; all OOB \(\hat\tau\) predictions are at this level (not person–video). Treatment \(W\) is workshop_Treatment (constant within respondent). We try clusters = ID (person, higher than the unit) so multiple party-cells per person are not treated as independent; if grf rejects that, forests are fit without clustering rather than coarsening units.

Covariates \(X\). Respondent-level covariates from labels_en, merged from df_w9_w10 via ID / id_w9, plus full sending-party dummies for the cell’s party (same coding as party_codes / party names). Party must enter \(X\) so \(\hat\tau(X)\) can differ by sending party (including AfD vs non-AfD). Rows with missing \(Y\), \(W\), or \(X\) are dropped (listwise). Covariates with very high missingness on this sample (notably structurally missing tiktok_potential_* for current users, and berufliche_stellung) are omitted from \(X\) before listwise deletion so the forest sample is not restricted to non-users; omissions are reported in the chunk output when forests are fit.

Reading the 4×4. Off-diagonal panels scatter OOB CATE estimates for one outcome against another (each point = one person×party unit). Positive correlation means units with more positive estimated effects on one outcome also tend to have more positive effects on the other. Of particular interest: whether more negative Like CATEs co-occur with more positive Report CATEs. Diagonal panels show the marginal distribution of each outcome’s CATE. A numeric correlation matrix is saved alongside the figure.

  x_resp_use <- setdiff(x_available, x_high_miss)
  x_use <- c(x_resp_use, party_x_cols)

  n_before <- nrow(df_cf)

  mutate(
    x_idx = match(var2, var_levels),
    y_idx = match(var1, var_levels)
  )

# Upper triangle: filled ellipses inside cells, scaled to avoid overlap
scale_in_cell <- 0.42
ellipse_upper <- cor_long |>
  filter(y_idx < x_idx) |>
  group_by(wave, treat, var1, var2, r, x_idx, y_idx, sign) |>
  do({
    rr <- .$r[1]
    S <- matrix(c(1, rr, rr, 1), nrow = 2)
    e <- as.data.frame(ellipse::ellipse(S, centre = c(0, 0), level = 0.68, npoints = 80))
    # normalize to unit box then scale to cell fraction
    maxabs <- max(max(abs(e$x)), max(abs(e$y)), 1e-9)
    e$x <- (e$x / maxabs) * scale_in_cell + .$x_idx[1]
    e$y <- (e$y / maxabs) * scale_in_cell + .$y_idx[1]
    e$wave <- .$wave[1]
    e$treat <- .$treat[1]
    e$var1 <- .$var1[1]
    e$var2 <- .$var2[1]
    e$sign <- .$sign[1]
    e$r <- rr
    e
  }) |>
  ungroup()

# Lower triangle: numeric labels
label_lower <- cor_long |>
